# Letakkan file ini di ROOT proyek (folder yang berisi 'public').
# ============================================================

<IfModule mod_rewrite.c>
    RewriteEngine On

    # --- 1. Arahkan semua request ke folder public/ ---
    RewriteRule ^(.*)$ public/$1 [L]

    # --- 2. Blokir akses langsung ke file sensitif ---
    RewriteRule ^\.env$ - [F,L]
    RewriteRule ^(?:\.git|vendor|storage|bootstrap|cache)/ - [F,L,NC]
</IfModule>

# --- 3. Cegah index direktori ---
Options -Indexes
